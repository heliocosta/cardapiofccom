{{/* cfc/img-src.html - returns a resized URL for remote images when Cloudflare Image Transformations are enabled. Args: dict "src" "width" . Enable with params.cfc_image_resize = true (CI sets env HUGOxPARAMSxCFC_IMAGE_RESIZE=true). Local/dev builds keep the original URL. */}} {{- $src := .src -}} {{- $w := mul (.width | default 40) 2 -}} {{- if and (eq (printf "%v" site.Params.cfc_image_resize) "true") (hasPrefix $src "http") -}} {{- printf "/cdn-cgi/image/width=%d,quality=80,format=auto/%s" $w $src -}} {{- else -}} {{- $src -}} {{- end -}}
