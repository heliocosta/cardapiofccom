Args: Page, ctx */}} {{- $p := .page -}} {{- $inLang := site.Language.Locale | default site.Language.Lang -}} {{- $name := $p.Title -}} {{- if and .ctx .ctx.title -}} {{- $name = .ctx.title -}} {{- end -}} {{- $desc := $p.Description | default $p.Params.description | default "" -}} {{- if and .ctx .ctx.desc -}} {{- $desc = .ctx.desc -}} {{- end -}} {{- $posts := slice -}} {{- range $p.Pages.ByDate.Reverse -}} {{- $entry := dict "@type" "BlogPosting" "headline" .Title "url" .Permalink -}} {{- if not .Date.IsZero -}} {{- $entry = merge $entry (dict "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00")) -}} {{- end -}} {{- $posts = $posts | append $entry -}} {{- end -}} {{- $node := dict "@type" "Blog" "@id" (printf "%s#blog" $p.Permalink) "name" $name "description" $desc "url" $p.Permalink "inLanguage" $inLang "publisher" (dict "@type" "Organization" "@id" "https://cardapiofc.com/#organization" "name" "CardápioFC") "blogPost" $posts -}} {{- return $node -}}
